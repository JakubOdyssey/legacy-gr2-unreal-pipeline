cmake_minimum_required(VERSION 3.16)
project(legacy_fbx_audit LANGUAGES C)
if(NOT UFBX_SOURCE_DIR)
  message(FATAL_ERROR "Use tools/build_reader.py or provide the reviewed UFBX_SOURCE_DIR")
endif()
add_executable(ufbx_audit tools/ufbx_audit.c "${UFBX_SOURCE_DIR}/ufbx.c")
target_include_directories(ufbx_audit PRIVATE "${UFBX_SOURCE_DIR}")
set_target_properties(ufbx_audit PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
if(NOT MSVC)
  target_link_libraries(ufbx_audit PRIVATE m)
endif()
